Overview

Wraps an array of Task objects running on a shared Reactor. The reactor is created on demand if not provided.

def spawn
  @tasks.each do |task|
    task.reactor = @reactor
    task.spawn
  end
  nil
ensure
  @spawned = true
end

def wait
  spawn unless @spawned
  @tasks.map(&:wait)
ensure
  @reactor.stop
end
