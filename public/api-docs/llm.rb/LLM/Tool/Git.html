def call(action:, arguments: [], timeout: 5)
  command = spawn(action:, arguments:)
  wait(command:, timeout:)
  {ok: command.success?, stdout: command.stdout, stderr: command.stderr}
rescue LLM::Interrupt
  command.kill! if command&.running?
  raise
end
